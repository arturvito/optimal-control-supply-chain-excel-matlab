function main_ga_minmax_restricao
    clear; clc; close all;

    %% 1. Configuração do Problema
    N = 60;
    
    % Demanda
    d_profile = zeros(1, N);
    d_profile(1:30)   = 20;
    d_profile(31:60)  = 10;

    % Número de Variáveis
    nvars = 3 * N;

    % Configuração GA
    IntCon = 1:nvars;
    
    % Limites
    lb = zeros(1, nvars);      
    ub = ones(1, nvars) * 400; 

    % População Inicial
    InitialPop = [];
    InitialPop = [InitialPop; [d_profile, d_profile, d_profile]]; 
    InitialPop = [InitialPop; ones(1, nvars)*100];
    InitialPop = [InitialPop; ones(1, nvars)*20];
    
    % Opções do GA
    options = optimoptions('ga', ...
        'Display', 'iter', ...
        'PopulationSize', 400, ...      
        'MaxGenerations', 400, ...      
        'InitialPopulationMatrix', InitialPop, ...
        'UseParallel', true, ...        
        'PlotFcns', {@gaplotbestf, @gaplotstopping});

    %% 2. Chamada do GA
    fprintf('Iniciando a Otimização MinMax \n');
    
    % Função Objetivo
    objective = @(x) custo_total(x, N, d_profile);
    [x_opt, fval, exitflag, output] = ga(objective, nvars, [], [], [], [], lb, ub, [], IntCon, options);

    %% 3. Visualização
    [u1_opt, u2_opt, u3_opt] = vetor_para_series(x_opt, N);
    [EA, ED, EV, VV] = simular_cadeia(u1_opt, u2_opt, u3_opt, d_profile, N);

    % save shooting60meses.mat  u1_opt u2_opt u3_opt EA ED EV VV 

    t = 1:N;
    figure('Name', 'Solucao MinMax com Restricao VV<=d', 'Color', 'w', 'Position', [100 100 1000 900]);
    
    % Plot 1: Referências
    figure(2)
    stairs(t, u1_opt, 'b', 'LineWidth', 2.5); hold on;
    stairs(t, u2_opt, 'r', 'LineWidth', 2.5);
    stairs(t, u3_opt, 'g', 'LineWidth', 2.5);
    yline(100, 'k:', 'Estado Inicial');
    legend('Ref. Atacadista', 'Ref. Distribuidor', 'Ref. Varejista', 'Location', 'best');
    title('Referências'); 
    ylabel('itens')
    xlabel('períodos')
    xticks([1:2:59])
    yticks([60:2:100])
    axis([0 60 60 100])
    grid on;
    exportgraphics(gcf, "controles.png", "Resolution",300);
    
    % Plot 2: Estoques
    figure(3)
    stairs(t, EA, 'b', 'LineWidth', 2.5); hold on; 
    stairs(t, ED, 'r', 'LineWidth', 2.5); 
    stairs(t, EV, 'g', 'LineWidth', 2.5);
    legend('Atacadista', 'Distribuidor', 'Varejista');
    title('Estoques Físicos'); 
    ylabel('Nível');
    xlabel('períodos')
    xticks([1:2:59])
    yticks([10:5:110])
    axis([0 60 10 110])
    grid on;
    exportgraphics(gcf, "estoques.png", "Resolution",300);
    
    % Plot 3: Tracking
    figure(4)
    stairs(t, d_profile, 'k', 'LineWidth', 2.5); hold on;
    stairs(t, VV, 'Color', [1 0.5 0], 'LineWidth', 2.5);
    legend('Demanda', 'Vendas Realizadas');
    title('Desempenho de Vendas'); 
    ylabel('itens')
    xlabel('períodos')
    xticks([1:2:59])
    yticks([-2:2:22])
    axis([0 60 -2 22])
    grid on;
    exportgraphics(gcf, "vendas.png", "Resolution",300);
end

%% --- Funções Auxiliares ---

function [u1, u2, u3] = vetor_para_series(x, N)
    if iscolumn(x), x = x'; end
    u1 = x(1:N);
    u2 = x(N+1:2*N);
    u3 = x(2*N+1:3*N);
end

function [EA_out, ED_out, EV_out, VV_out] = simular_cadeia(u1_vec, u2_vec, u3_vec, d_vec, N)
    k1_p = 1; k1_i = 0;
    k2_p = 1; k2_i = 0;
    k3_p = 1; k3_i = 0;

    Ts = 1;
    sat_limit = 2000; 

    EA  = 100; ED  = 100; EV  = 100;
    PEA = 100; PED = 100; PEV = 100;
    PRP = 0;   PRA = 0;   PRD = 0;
    PPA = 0;   PPD = 0;   PPV = 0;
    VP  = 0;   VA  = 0;   VD  = 0;
    IA  = 0;   ID  = 0;   IV  = 0;

    EA_out = zeros(1, N); ED_out = zeros(1, N); EV_out = zeros(1, N); VV_out = zeros(1, N);

    for k = 1:N
        u1 = u1_vec(k); u2 = u2_vec(k); u3 = u3_vec(k); d = d_vec(k);

        mEA = min(EA, PPA);
        mED = min(ED, PPD);
        mEV = min(EV, PPV); 

        EA_n = EA + VP - mEA;
        ED_n = ED + VA - mED;
        EV_n = EV + VD - mEV;

        PEA_n = PEA + PRP - mEA;
        PED_n = PED + mEA - mED;
        PEV_n = PEV + mED - mEV;

        cmd_PRP = IA + k1_p*(u1 - PEA - PRP + mEA);
        cmd_PRA = ID + k2_p*(u2 - PED - mEA + mED);
        cmd_PRD = IV + k3_p*(u3 - PEV - mED + mEV);

        PRP_n = max(0, min(sat_limit, cmd_PRP));
        PRA_n = max(0, min(sat_limit, cmd_PRA));
        PRD_n = max(0, min(sat_limit, cmd_PRD));

        PPA_n = PPA - mEA + PRA_n; 
        PPD_n = PPD - mED + PRD_n; 
        PPV_n = PPV - mEV + d;

        erro1 = (u1 - PEA - PRP + mEA);
        erro2 = (u2 - PED - mEA + mED);
        erro3 = (u3 - PEV - mED + mEV);
        
        IA_n = max(-sat_limit, min(sat_limit, IA + k1_i*Ts*erro1));
        ID_n = max(-sat_limit, min(sat_limit, ID + k2_i*Ts*erro2));
        IV_n = max(-sat_limit, min(sat_limit, IV + k3_i*Ts*erro3));

        PPA_n = max(0, min(sat_limit, PPA_n));
        PPD_n = max(0, min(sat_limit, PPD_n));
        PPV_n = max(0, min(sat_limit, PPV_n));

        VP_n = PRP_n; VA_n = mEA; VD_n = mED;   

        EA_out(k) = EA_n; ED_out(k) = ED_n; EV_out(k) = EV_n; VV_out(k) = mEV;

        EA=EA_n; ED=ED_n; EV=EV_n; PEA=PEA_n; PED=PED_n; PEV=PEV_n;
        PRP=PRP_n; PRA=PRA_n; PRD=PRD_n; PPA=PPA_n; PPD=PPD_n; PPV=PPV_n;
        VP=VP_n; VA=VA_n; VD=VD_n; IA=IA_n; ID=ID_n; IV=IV_n;
    end
end

function J = custo_total(x, N, d)
    if iscolumn(x), x = x'; end
    
    [u1, u2, u3] = vetor_para_series(x, N);
    [EA, ED, EV, VV] = simular_cadeia(u1, u2, u3, d, N);
    
    % Pesos
    w_tracking = 500;  
    w_estoque  = 250;     
    w_penalty_neg = 10000; % Estoques negativos
    w_penalty_vv  = 10000; % vv <= d
    w_smooth   = 0.1;
    
    J_track = sum((VV - d).^2);
    
    % Custo MinMax dos Estoques
    estoques_matrix = [EA; ED; EV];
    max_inst_stock = max(estoques_matrix, [], 1);
    J_stock = sum(max_inst_stock);
    
    % Penalidades
    penalty_neg = sum(max(0, -EA).^2) + sum(max(0, -ED).^2) + sum(max(0, -EV).^2);
    
    % Se VV > d, adiciona custo, caso contrário, custo é 0.
    penalty_vv = sum(max(0, VV - d).^2);
    
    J_smooth = sum(abs(diff(u1))) + sum(abs(diff(u2))) + sum(abs(diff(u3)));
    
    J = w_tracking * J_track + ...
        w_estoque  * J_stock + ...
        w_penalty_neg * penalty_neg + ...
        w_penalty_vv  * penalty_vv + ...
        w_smooth   * J_smooth;
end
